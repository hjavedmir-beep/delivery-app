async function searchPostcodeAddresses() { const postcode = document.getElementById('busPostcode').value.trim(); if (!postcode) { alert('Please enter a valid postcode first.'); return; } try { // یہ ہے Postcodes.io کی پبلک API جو بغیر کسی کی کے کام کرتی ہے const res = await fetch(`https://api.postcodes.io/postcodes/${encodeURIComponent(postcode)}`); const data = await res.json(); const dropdownContainer = document.getElementById('addressSelectContainer'); const dropdown = document.getElementById('busAddressDropdown'); dropdown.innerHTML = ''; if (data.status === 200 && data.result) { const resObj = data.result; dropdownContainer.classList.remove('hidden'); // یہاں سے ہم لیٹیٹیوڈ اور لونگیٹیوڈ حاصل کرتے ہیں dropdown.dataset.lat = resObj.latitude; dropdown.dataset.lng = resObj.longitude; let locationName = `${resObj.admin_ward || resObj.parish || ''}, ${resObj.admin_district}, ${postcode}`; dropdown.innerHTML += ``; document.getElementById('busAddress').value = `${resObj.admin_district}, ${postcode}`; } else { dropdownContainer.classList.add('hidden'); alert('Postcode not found in UK database. Please type your street address manually.'); } } catch (e) { console.error("Error looking up postcode", e); alert('Error connecting to postcode server. You can type the address manually.'); } }
