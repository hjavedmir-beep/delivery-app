async function searchPostcodeAddresses() { const postcode = document.getElementById('busPostcode').value.trim(); if (!postcode) { alert('Please enter a valid postcode first.'); return; } try { // ہم یہاں OpenStreetMap (Nominatim) API استعمال کر رہے ہیں جو پوسٹ کوڈ سے پورا علاقہ اور ایڈریس ڈھونڈ کر دیتی ہے const res = await fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(postcode)}&addressdetails=1`); const data = await res.json(); if (data && data.length > 0) { const firstResult = data[0]; // کوآرڈینیٹس سیٹ کرنا تاکہ میپ پر بالکل صحیح پن لگ سکے document.getElementById('busAddress').dataset.lat = firstResult.lat; document.getElementById('busAddress').dataset.lng = firstResult.lon; // علاقے اور شہر کا مکمل نام ایڈریس کے خانے میں خود بخود آ جائے گا const displayName = firstResult.display_name; document.getElementById('busAddress').value = displayName; alert('Location found: ' + firstResult.name || 'Area loaded successfully!'); } else { alert('Postcode not found. Please type your street address manually.'); } } catch (e) { console.error("Error looking up postcode", e); alert('Error connecting to location server. You can type the address manually.'); } }
